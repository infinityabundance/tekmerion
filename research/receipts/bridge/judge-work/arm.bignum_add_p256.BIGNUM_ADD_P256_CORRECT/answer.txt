(
(* ========================================================================= *)
(* Reading and writing DIMACS ".cnf" files and converting to and from HOL.   *)
(*                                                                           *)
(* This does not rigorously enforce many of the syntactic restrictions and   *)
(* in particular it completely ignores the "p" line (or many "p" lines)      *)
(* specifying the number of variables and clauses. It also doesn't do any    *)
(* analysis of the clauses to rule out tautologies etc.                      *)
(* ========================================================================= *)

(* ------------------------------------------------------------------------- *)
(* Equivalents of "conjuncts" and "length o conjuncts" but tail recursive.   *)
(* ------------------------------------------------------------------------- *)

let list_conjuncts =
  let rec aux acc tm =
    match tm with
      Comb(Comb(Const("/\\",_),l),t) -> aux (l::acc) t
    | _ -> tm::acc in
  fun tm -> rev(aux [] tm)
in
let rec count_conjuncts n tm =
  match tm with
    Comb(Comb(Const("/\\",_),_),t) -> count_conjuncts (n+1) t
  | _ -> n
in
(* ------------------------------------------------------------------------- *)
(* Parsing and processing.                                                   *)
(* ------------------------------------------------------------------------- *)

let stripcomments s =
  match String.index_opt s 'c' with
    Some i -> String.sub s 0 i
  | None -> s
in
let process_lines f p fname =
  let fd =
    try open_in fname
    with Sys_error _ -> failwith("read_dimacs: can't open "^fname) in
  let rec suck_lines acc =
    let l = try [input_line fd] with End_of_file -> [] in
    if l = [] then rev_itlist (@) acc [] else
    let y = f(hd l) in
    suck_lines (if p y then y::acc else acc) in
  let data = suck_lines [] in
  (close_in fd; data)
in
let rec parse_clauses acc cur l =
  match l with
    ""::t -> parse_clauses  acc cur t
  | "0"::t -> parse_clauses (rev cur::acc) [] t
  | n::t -> parse_clauses acc (int_of_string n::cur) t
  | [] -> rev acc
in
(* ------------------------------------------------------------------------- *)
(* Reading and writing betweeen DIMACS file and int list list in OCaml       *)
(* ------------------------------------------------------------------------- *)

let read_dimacs fname =
  let l = process_lines
           (String.split_on_char ' ' o
            String.map (function '\t' -> ' ' | c -> c) o stripcomments)
           (function "p"::_ -> false | _ -> true) fname in
  parse_clauses [] [] l
in
let write_dimacs =
  let string_of_clause cl =
    itlist (fun t s -> string_of_int t ^ " " ^ s) cl "0\n" in
  fun fname cls ->
    let fd =
      try open_out fname
      with Sys_error _ -> failwith("write_dimacs: can't open "^fname) in
    let n = rev_itlist (rev_itlist (max o abs)) cls 0
    and m = rev_itlist (fun _ n -> n + 1) cls 0 in
    (output_string fd ("p cnf "^string_of_int n^" "^string_of_int m^"\n");
     do_list (output_string fd o string_of_clause) cls;
     close_out fd)
in
(* ------------------------------------------------------------------------- *)
(* Translation between HOL terms and int list list, with variable mappings.  *)
(* ------------------------------------------------------------------------- *)

let clauses_of_term vmap tm =
  let cnf_of_literal t =
    match t with
      Comb(Const("~",_),s) -> -(vmap s)
    | _ -> vmap t in
  map (map cnf_of_literal o disjuncts) (list_conjuncts tm)
in
let term_of_literal amap n =
   if n < 0 then mk_neg(amap (-n)) else amap n
in
let term_of_clause =
  let false_tm = `F` in
  fun amap ->
    let rec aux tm l =
      match l with
        h::t -> aux (mk_disj(term_of_literal amap h,tm)) t
      | [] -> tm in
    fun l ->
      match rev l with
        [] -> false_tm
      | h::t -> aux (term_of_literal amap h) t
in
let term_of_clauses =
  let true_tm = `T` in
  fun amap ->
    let rec aux tm l =
      match l with
        h::t -> aux (mk_conj(term_of_clause amap h,tm)) t
      | [] -> tm in
    fun l ->
      match rev l with
        [] -> true_tm
      | h::t -> aux (term_of_clause amap h) t
in
(* ------------------------------------------------------------------------- *)
(* Composite function not directly creating the int list list and picking a  *)
(* default numbering of the atoms from the call to "atoms".                  *)
(* ------------------------------------------------------------------------- *)

let dimacs_of_hol fname tm =
  let m = count_conjuncts 1 tm in
  let ats = atoms tm in
  let n = length ats in
  let _,vmap =
    rev_itlist (fun a (n,f) -> (n+1,(a |-> n) f)) ats (1,undefined) in
  let cnf_of_literal t =
    match t with
      Comb(Const("~",_),s) -> -(apply vmap s)
    | _ -> apply vmap t in
  let string_of_clause cl =
   itlist (fun t s -> string_of_int(cnf_of_literal t) ^ " " ^ s)
          (disjuncts cl) "0\n" in
  let fd =
    try open_out fname
    with Sys_error _ -> failwith("dimacs_of_hol: can't open "^fname) in
  (output_string fd ("p cnf "^string_of_int n^" "^string_of_int m^"\n");
   do_list (output_string fd o string_of_clause) (list_conjuncts tm);
   close_out fd)
in
(* ========================================================================= *)
(* Tools to put formulas in definitional (Tseitin) CNF form for SAT solvers. *)
(* The temporary variables are implicit for actual proof reconstruction; all *)
(* steps simply use the corresponding expansion directly.                    *)
(* ========================================================================= *)

let exploit_conjunctive_structure = ref true
in
(* ------------------------------------------------------------------------- *)
(* Split up a theorem according to conjuncts, in a general sense.            *)
(* ------------------------------------------------------------------------- *)

let GCONJUNCTS =
  let [pth_ni1; pth_ni2; pth_no1; pth_no2; pth_an1; pth_an2; pth_nn] =
    (map UNDISCH_ALL o CONJUNCTS o TAUT)
    `(~(p ==> q) ==> p) /\
     (~(p ==> q) ==> ~q) /\
     (~(p \/ q) ==> ~p) /\
     (~(p \/ q) ==> ~q) /\
     (p /\ q ==> p) /\
     (p /\ q ==> q) /\
     (~ ~p ==> p)` in
  let p_tm = concl pth_an1 and q_tm = concl pth_an2 in
  let rec GCONJUNCTS th acc =
    match (concl th) with
      Comb(Const("~",_),Comb(Comb(Const("==>",_),p),q)) ->
       GCONJUNCTS (PROVE_HYP th (INST [p,p_tm; q,q_tm] pth_ni1))
                  (GCONJUNCTS (PROVE_HYP th (INST [p,p_tm; q,q_tm] pth_ni2))
                              acc)
    | Comb(Const("~",_),Comb(Comb(Const("\\/",_),p),q)) ->
       GCONJUNCTS (PROVE_HYP th (INST [p,p_tm; q,q_tm] pth_no1))
                  (GCONJUNCTS (PROVE_HYP th (INST [p,p_tm; q,q_tm] pth_no2))
                              acc)
    | Comb(Comb(Const("/\\",_),p),q) ->
       GCONJUNCTS (PROVE_HYP th (INST [p,p_tm; q,q_tm] pth_an1))
                  (GCONJUNCTS (PROVE_HYP th (INST [p,p_tm; q,q_tm] pth_an2))
                              acc)
    | Comb(Const("~",_),Comb(Const("~",_),p)) ->
          GCONJUNCTS (PROVE_HYP th (INST [p,p_tm] pth_nn)) acc
    | _ -> th::acc in
  fun th -> GCONJUNCTS th []
in
(* ------------------------------------------------------------------------- *)
(* Generate fresh variable names (could just use genvars).                   *)
(* ------------------------------------------------------------------------- *)

let propvar i = mk_var("x"^string_of_int i,bool_ty)
in
(* ------------------------------------------------------------------------- *)
(* Set up the basic definitional arrangement.                                *)
(* ------------------------------------------------------------------------- *)

let rec localdefs tm (n,defs,lfn) =
  if is_neg tm then
    let n1,v1,defs1,lfn1 = localdefs (rand tm) (n,defs,lfn) in
    let tm' = mk_neg v1 in
    try (n1,apply defs1 tm',defs1,lfn1) with Failure _ ->
    let n2 = n1 + 1 in
    let v2 = propvar n2 in
    n2,v2,(tm' |-> v2) defs1,(v2 |-> tm) lfn1
  else if is_conj tm || is_disj tm || is_imp tm || is_iff tm then
    let n1,v1,defs1,lfn1 = localdefs (lhand tm) (n,defs,lfn) in
    let n2,v2,defs2,lfn2 = localdefs (rand tm) (n1,defs1,lfn1) in
    let tm' = mk_comb(mk_comb(rator(rator tm),v1),v2) in
    try (n2,apply defs2 tm',defs2,lfn2) with Failure _ ->
    let n3 = n2 + 1 in
    let v3 = propvar n3 in
    n3,v3,(tm' |-> v3) defs2,(v3 |-> tm) lfn2
  else try (n,apply defs tm,defs,lfn) with Failure _ ->
       let n1 = n + 1 in
       let v1 = propvar n1 in
       n1,v1,(tm |-> v1) defs,(v1 |-> tm) lfn
in
(* ------------------------------------------------------------------------- *)
(* Just translate to fresh variables, but otherwise leave unchanged.         *)
(* ------------------------------------------------------------------------- *)

let rec transvar (n,tm,vdefs,lfn) =
  if is_neg tm then
    let n1,tm1,vdefs1,lfn1 = transvar (n,rand tm,vdefs,lfn) in
    n1,mk_comb(rator tm,tm1),vdefs1,lfn1
  else if is_conj tm || is_disj tm || is_imp tm || is_iff tm then
    let n1,tm1,vdefs1,lfn1 = transvar (n,lhand tm,vdefs,lfn) in
    let n2,tm2,vdefs2,lfn2 = transvar (n1,rand tm,vdefs1,lfn1) in
    n2,mk_comb(mk_comb(rator(rator tm),tm1),tm2),vdefs2,lfn2
  else try n,apply vdefs tm,vdefs,lfn with Failure _ ->
       let n1 = n + 1 in
       let v1 = propvar n1 in
       n1,v1,(tm |-> v1) vdefs,(v1 |-> tm) lfn
in
(* ------------------------------------------------------------------------- *)
(* Determine if a term is (strictly) clausal, i.e. is a conjunction of       *)
(* disjunctions of literals, all of which are right-associated.              *)
(* ------------------------------------------------------------------------- *)

let is_atom tm =
  match tm with
    Comb(Comb(Const("/\\",_),l),r)
  | Comb(Comb(Const("\\/",_),l),r)
  | Comb(Comb(Const("==>",_),l),r)
  | Comb(Comb(Const("=",Tyapp("fun",[Tyapp("bool",[]);_])),l),r)
        -> false
  | Comb(Const("~",_),l) -> false
  | _ -> true
in
let is_literal tm =
  match tm with
    Comb(Const("~",_),t) -> is_atom t
  | _ -> is_atom tm
in
let rec is_clausal tm =
  match tm with
    Comb(Comb(Const("\\/",_),l),r) -> is_literal l && is_clausal r
  | _ -> is_literal tm
in
let rec is_strictcnf tm =
  match tm with
    Comb(Comb(Const("/\\",_),l),r) -> is_clausal l && is_strictcnf r
  | _ -> is_clausal tm
in
(* ------------------------------------------------------------------------- *)
(* Now do the definitional arrangement but not wastefully at the top.        *)
(* ------------------------------------------------------------------------- *)

let definitionalize =
  let transform_imp =
    let pth = TAUT `(p ==> q) <=> ~p \/ q` in
    let ptm = rand(concl pth) in
    let p_tm = rand(lhand ptm) and q_tm = rand ptm in
    fun th -> let ip,q = dest_comb(concl th) in
              let p = rand ip in
              EQ_MP (INST [p,p_tm; q,q_tm] pth) th
  and transform_iff_1 =
    let pth = UNDISCH(TAUT `(p <=> q) ==> (p \/ ~q)`) in
    let ptm = concl pth in
    let p_tm = lhand ptm and q_tm = rand(rand ptm) in
    fun th -> let ip,q = dest_comb(concl th) in
              let p = rand ip in
              PROVE_HYP th (INST [p,p_tm; q,q_tm] pth)
  and transform_iff_2 =
    let pth = UNDISCH(TAUT `(p <=> q) ==> (~p \/ q)`) in
    let ptm = concl pth in
    let p_tm = rand(lhand ptm) and q_tm = rand ptm in
    fun th -> let ip,q = dest_comb(concl th) in
              let p = rand ip in
              PROVE_HYP th (INST [p,p_tm; q,q_tm] pth) in
  let definitionalize th (n,tops,defs,lfn) =
    let t = concl th in
    if is_clausal t then
      let n',v,defs',lfn' = transvar (n,t,defs,lfn) in
      (n',(v,th)::tops,defs',lfn')
    else if is_neg t then
       let n1,v1,defs1,lfn1 = localdefs (rand t) (n,defs,lfn) in
       (n1,(mk_neg v1,th)::tops,defs1,lfn1)
    else if is_disj t then
      let n1,v1,defs1,lfn1 = localdefs (lhand t) (n,defs,lfn) in
      let n2,v2,defs2,lfn2 = localdefs (rand t) (n1,defs1,lfn1) in
      (n2,(mk_disj(v1,v2),th)::tops,defs2,lfn2)
    else if is_imp t then
      let n1,v1,defs1,lfn1 = localdefs (lhand t) (n,defs,lfn) in
      let n2,v2,defs2,lfn2 = localdefs (rand t) (n1,defs1,lfn1) in
      (n2,(mk_disj(mk_neg v1,v2),transform_imp th)::tops,defs2,lfn2)
    else if is_iff t then
      let n1,v1,defs1,lfn1 = localdefs (lhand t) (n,defs,lfn) in
      let n2,v2,defs2,lfn2 = localdefs (rand t) (n1,defs1,lfn1) in
      (n2,(mk_disj(v1,mk_neg v2),transform_iff_1 th)::
          (mk_disj(mk_neg v1,v2),transform_iff_2 th)::tops,defs2,lfn2)
    else
      let n',v,defs',lfn' = localdefs t (n,defs,lfn) in
      (n',(v,th)::tops,defs',lfn') in
  definitionalize
in
(* ========================================================================= *)
(* Translate .lrat proof file into HOL inferences and prove a .cnf formula.  *)
(*                                                                           *)
(* This is designed for the .lrat proofs produced by Cadical; those do not   *)
(* use proper RAT lemmas, and have a particularly clean and simple syntax.   *)
(* This might not work for the .lrat files produced by other provers.        *)
(*                                                                           *)
(* The first argument "amap" is the mapping from .cnf variable numbers to    *)
(* HOL propositions, e.g. 123 |-> `p123:bool`.                               *)
(*                                                                           *)
(* For the _GEN functions there are two filename arguments, one for the CNF  *)
(* problem file and the other for the corresponding LRAT proof file. For the *)
(* non-GEN forms there is a single argument that is the root "filename"      *)
(* for the corresponding CNF file "filename.cnf" and LRAT "filename.lrat".   *)
(*                                                                           *)
(* In all cases, non-absolute filenames are interpreted relative to the      *)
(* directory pointed at by the reference "problem_dir".                      *)
(* ========================================================================= *)

let problem_dir = ref "/tmp"
in
let LRAT_PROVE_GEN =
  let ptm = `p:bool` and qtm = `q:bool` in
  let orth = REFL(rator(rator(mk_disj(ptm,ptm)))) in
  let number_clauses =
    let rec aux f n (cl,th) =
      match cl with
        [c] -> (n |-> (c,th)) f
      | c::cs ->
          let th1,th2 = CONJ_PAIR th in
          aux ((n |-> (c,th1)) f) (n + 1) (cs,th2)
      | _ -> failwith "number_clauses: there are no clauses" in
    aux undefined 1 in
  let DEDRUP_RULE =
    let pth = SPEC ptm (TAUT `!p. (~p ==> F) <=> p`) in
    fun th -> EQ_MP (INST [rand(lhand(concl th)),ptm] pth) th in
  let ORIFY (c,th) uths = (-c |-> (th,MK_COMB(orth,th))) uths in
  let EQUIFY =
    let nth = SPEC ptm (TAUT `!p. ~p <=> (p <=> F)`)
    and pth = SPEC ptm (TAUT `!p. p <=> (~p <=> F)`) in
    fun (c,th) uths ->
      let tm = concl th in
      let th' =
        if c < 0 then EQ_MP (INST [rand tm,ptm] nth) th
        else EQ_MP (INST [tm,ptm] pth) th in
      let th'' = MK_COMB(orth,th') in
      (-c |-> (th',th'')) uths in
    let upth_l = SPEC ptm (TAUT `!p. F \/ p <=> p`)
    and upth_r = SPEC ptm (TAUT `!p. p \/ F <=> p`)
    and upth_b = TAUT `F \/ F <=> F` in
  fun amap cnfname lratname ->
    let adjust s =
      if Filename.is_relative s
      then Filename.concat (!problem_dir) s else s in
    let probfile = adjust cnfname and lratfile = adjust lratname in
    let prb = read_dimacs probfile in
    let ctm = term_of_clauses amap prb in
    let nvars = rev_itlist (rev_itlist (max o abs)) prb 0 in
    let upth_l_pos = Array.init (nvars+1)
      (fun v -> if v = 0 then upth_b else INST [amap v,ptm] upth_l)
    and upth_l_neg = Array.init (nvars+1)
      (fun v -> if v = 0 then upth_b else INST [mk_neg(amap v),ptm] upth_l)
    and upth_r_pos = Array.init (nvars+1)
      (fun v -> if v = 0 then upth_b else INST [amap v,ptm] upth_r)
    and upth_r_neg = Array.init (nvars+1)
      (fun v -> if v = 0 then upth_b else INST [mk_neg(amap v),ptm] upth_r) in
    let upth_t_pos = Array.init (nvars+1)
        (fun v -> REFL(rator(lhand(concl(Array.get upth_r_pos v)))))
    and upth_t_neg = Array.init (nvars+1)
        (fun v -> REFL(rator(lhand(concl(Array.get upth_r_neg v))))) in
    let UNITPROP =
      fun uths ->
        let rec conv (cl,tm) =
            match (cl,tm) with
              [c],_ -> (try 0,fst(apply uths c) with Failure _ -> c,REFL tm)
            | (c::cs,Comb(Comb(_,l),r)) ->
                (let rl,rth = conv(cs,r) in
                 try let lth,lth' = apply uths c in
                     let th = MK_COMB(lth',rth) in
                     rl,TRANS th (if rl < 0 then Array.get upth_l_neg (-rl)
                                  else Array.get upth_l_pos rl)
                 with Failure _ ->
                   c,
                (if c < 0 then
                   TRANS (MK_COMB(Array.get upth_t_neg (-c),rth))
                         (Array.get upth_r_neg (-c))
                 else
                   TRANS (MK_COMB(Array.get upth_t_pos c,rth))
                         (Array.get upth_r_pos c)))
            | _ -> failwith "UNITPROP: Sanity check" in
      fun (cl,th) -> let d,th' = conv(cl,concl th) in d,EQ_MP th' th in
    let UNITPROPS cls =
      let rec rule uths pf =
        match pf with
          c::ops -> let p,th = UNITPROP uths (apply cls c) in
                    if p = 0 then th else rule (EQUIFY (p,th) uths) ops
        | [] -> failwith "UNITPROPS: did not reach a contradiction" in
      fun dths -> rule (rev_itlist ORIFY dths undefined) in
    let NEGATE_CLAUSE =
      let pth_lor = UNDISCH(SPECL[ptm;qtm]
       (TAUT `!p q. ~(p \/ q) ==> (p <=> F)`))
      and pth_ror = UNDISCH(SPECL[ptm;qtm] (TAUT `!p q. ~(p \/ q) ==> ~q`))
      and pth_ide = SPEC ptm (TAUT `!p. ~p <=> (p <=> F)`) in
      let nrule (c,th) =
        let Comb(_,t) = concl th in
        -c,EQ_MP (INST [t,ptm] pth_ide) th in
      let rec rule (cl,th) =
        match (cl,concl th) with
          [c],_ -> [nrule(c,th)]
        | (c::cs,Comb(_,Comb(Comb(_,l),r))) ->
            let iptho = INST [l,ptm; r,qtm] in
            let th1 = EQ_MP (DEDUCT_ANTISYM_RULE th (iptho pth_lor)) th
            and th2 = EQ_MP (DEDUCT_ANTISYM_RULE th (iptho pth_ror)) th in
            (-c,th1)::rule (cs,th2)
        | [],_ -> []
        | _ -> failwith "NEGATE_CLAUSE: sanity check" in
      rule in
      let LRAT_ADD (n,cl,pf) cls =
        let ctm = term_of_clause amap cl in
        let ctm' = mk_neg ctm in
        let dths = NEGATE_CLAUSE (cl,ASSUME ctm') in
        let fth = UNITPROPS cls dths pf in
        let th = DEDRUP_RULE(DISCH ctm' fth) in
        (n |-> (cl,th)) cls in
      let rec PROCESS_LRAT fd m cls =
        match (try Some(input_line fd) with End_of_file -> None) with
         Some s ->
           (match String.split_on_char ' ' s with
             ns::"d"::dnums ->
                  let n = int_of_string ns
                  and cls' = rev_itlist (undefine o int_of_string) dnums cls in
                  PROCESS_LRAT fd n cls'
            | ns::clap ->
                  let n = int_of_string ns
                  and clans = map int_of_string clap in
                  let i = index 0 clans in
                  let c,u = chop_list i clans in
                  let cls' = LRAT_ADD (n,c,butlast(tl u)) cls in
                  PROCESS_LRAT fd n cls'
            | _ -> failwith "PROCESS_LRAT: Can't parse input line")
        | None -> apply cls m in
    let cls = number_clauses(prb,ASSUME ctm) in
    let fd =
      try open_in lratfile
      with Sys_error _ -> failwith("LRAT_PROVE: can't open "^lratfile) in
    let _,fth = PROCESS_LRAT fd 0 cls in
    let _ = close_in fd in
    NOT_INTRO(DISCH ctm fth)
in
let LRAT_PROVE amap basename =
  LRAT_PROVE_GEN amap (basename^".cnf") (basename^".lrat")
in
let LRAT_REFUTE_GEN amap cnfname lratname =
    UNDISCH(NOT_ELIM(LRAT_PROVE_GEN amap cnfname lratname))
in
let LRAT_REFUTE amap basename =
   LRAT_REFUTE_GEN amap (basename^".cnf") (basename^".lrat")
in
(* ========================================================================= *)
(* csat_lib.ml -- certified-SAT close for HOL Light word goals.               *)
(*                                                                           *)
(* Connects the missing BITBLAST_THEN -> CADICAL_PROVE edge.  The pinned     *)
(* Examples/bitblast.ml defines BITBLAST_CADICAL only inside a (**** ...     *)
(* ****) block, so it is never loaded.  This file supplies it.               *)
(*                                                                           *)
(* Kernel trust: every SAT refutation is replayed by LRAT_REFUTE_GEN inside  *)
(* the HOL Light kernel (Cadical/ldrup.ml).  No solver certificate is trusted *)
(* and no compiler enters the trusted base.                                  *)
(*                                                                           *)
(* CaDiCaL 2.1.2 emits LRAT text directly with --no-binary --lrat; the       *)
(* lrat-trim post-pass used by Cadical/cadical.ml is therefore optional.     *)
(* CSAT_PROVE is CADICAL_PROVE with that native-LRAT refuter substituted.    *)
(* ========================================================================= *)

(* Unix (wall-clock) is loaded by the driver before this file is loadt-ed. *)
let csat_now () = Unix.gettimeofday ()
in
let csat_debug = ref false
in
(* --- native-LRAT refuter ------------------------------------------------- *)

let CSAT_REFUTE_GEN amap tm =
  let cnfname = Filename.temp_file "csathol" ".cnf"
  and lratname = Filename.temp_file "csathol" ".lrat"
  and logname = Filename.temp_file "csathol" ".log" in
  let _ = dimacs_of_hol cnfname tm in
  let command =
    "(cadical --no-binary --lrat "^cnfname^" "^lratname^") >"^logname in
  let rc = Sys.command command in
  if rc <> 20 then
    (ignore (Sys.command ("rm -f "^cnfname^" "^lratname^" "^logname));
     failwith ("CSAT: cadical did not report UNSAT, exit "^string_of_int rc));
  let res = LRAT_REFUTE_GEN amap cnfname lratname in
  let _ = if !csat_debug then ()
          else ignore (Sys.command
                 ("rm -f "^cnfname^" "^lratname^" "^logname)) in
  res
in
let CSAT_REFUTE =
  let false_tm = `F` in
  fun tm ->
    let aar = Array.of_list (false_tm::atoms tm) in
    let th = CSAT_REFUTE_GEN (Array.get aar) tm in
    if hyp th = [tm] && concl th = false_tm then th
    else failwith "CSAT_REFUTE: Sanity check failure"
in
(* --- definitional-CNF prover (mirrors CADICAL_PROVE) --------------------- *)

let CSAT_PROVE =
  let false_tm = `F`
  and p_tm = `p:bool` and q_tm = `q:bool`
  and pth_triv = TAUT `(~p <=> F) <=> p`
  and pth_main = UNDISCH_ALL(TAUT `(~p <=> q) ==> (q ==> F) ==> p`) in
  let presimp_conv = GEN_REWRITE_CONV DEPTH_CONV
    [NOT_CLAUSES; AND_CLAUSES; OR_CLAUSES; IMP_CLAUSES; EQ_CLAUSES]
  and triv_rule p th = EQ_MP(INST [p,p_tm] pth_triv) th
  and main_rule p q sth th =
    itlist PROVE_HYP [sth; DISCH_ALL th] (INST [p,p_tm; q,q_tm] pth_main) in
  fun tm ->
    if is_neg tm && is_strictcnf (rand tm) then
      NOT_INTRO(DISCH_ALL(CSAT_REFUTE(rand tm)))
    else if is_imp tm && rand tm = false_tm && is_strictcnf (lhand tm) then
      DISCH_ALL(CSAT_REFUTE(lhand tm))
    else
      let sth = presimp_conv(mk_neg tm) in
      let stm = rand(concl sth) in
      if stm = false_tm then triv_rule tm sth else
      let th = ASSUME stm in
      let ths = if !exploit_conjunctive_structure then GCONJUNCTS th
                else [th] in
      let n,tops,defs,lfn =
        itlist definitionalize ths (-1,[],undefined,undefined) in
      let defg = foldl (fun a t nv -> (t,nv)::a) [] defs in
      let mdefs = filter (fun (r,_) -> not (is_atom r)) defg in
      let eqs = map (fun (r,l) -> mk_iff(l,r)) mdefs in
      let clausify eq cls =
        let fvs = frees eq and eth = (NNFC_CONV THENC CNF_CONV) eq in
        let tth = INST (map (fun v -> apply lfn v,v) fvs) eth in
        let xth = ADD_ASSUM stm (EQ_MP tth (REFL(apply lfn (lhand eq)))) in
        zip (conjuncts(rand(concl eth))) (CONJUNCTS xth) @ cls in
      let all_clauses = itlist clausify eqs tops in
      let ntdcnf = list_mk_conj (map fst all_clauses) in
      let expat = map (apply lfn) (atoms ntdcnf) in
      let rar = Array.of_list(false_tm::expat) in
      let cth = CSAT_REFUTE_GEN (Array.get rar) ntdcnf in
      let dth = PROVE_HYP (end_itlist CONJ (map snd all_clauses)) cth in
      let fth = main_rule tm stm sth dth in
      if concl fth = tm && hyp fth = [] then fth
      else failwith "CSAT_PROVE: Sanity check"
in
let CSAT_TAC = BITBLAST_THEN (CONV_TAC o K CSAT_PROVE)
in
(* --- attempt a goal and report ------------------------------------------- *)

(* A goal (asl,w) becomes the theorem |- asl ==> w, closed with the
   certified-SAT tactic.  Returns (closed, seconds, message, theorem-term). *)
let csat_list_mk_imp (l,t) = itlist (fun a b -> mk_imp(a,b)) l t
in
let csat_attempt_goal (asl,w) =
  let tm = if asl = [] then w else
    csat_list_mk_imp (map (fun (_,th) -> concl th) asl, w) in
  let t0 = csat_now () in
  let closed,msg =
    try
      let th = prove(tm, REPEAT STRIP_TAC THEN CSAT_TAC) in
      (hyp th = [] && concl th = tm, "")
    with Failure s -> (false, s)
       | _ -> (false, "unknown exception") in
  (closed, csat_now () -. t0, msg, tm)
in
let report_attempt tag (closed,dt,msg,_) =
  if closed then
    Printf.printf "CSAT_RESULT %s closed=1 seconds=%.3f\n" tag dt
  else
    let s = String.concat " " (String.split_on_char '\n' msg) in
    Printf.printf "CSAT_RESULT %s closed=0 seconds=%.3f reason=%s\n" tag dt s
in
(* ========================================================================= *)
(* bridge.ml -- generic bridge lemmas from a num/MOD (or int/rem, real/&)    *)
(* specification to a word-level circuit statement that certified SAT can    *)
(* discharge.                                                                *)
(*                                                                           *)
(* Parameterisation: the modulus `p` is an argument of every lemma, the limb *)
(* width (64) enters only through the numerals that appear in the shifts,    *)
(* and the number of limbs enters only through the nesting depth of the      *)
(* joins.  There is no P-256 constant, no P-256 limb list and no transposed  *)
(* human tactic fragment in this file: the arithmetic below is the general   *)
(* statement that `(m + n) MOD p`, `(&m - &n) rem &p` and `(p - n) MOD p`    *)
(* are what the machine's written residue is, for any `p`.                   *)
(*                                                                           *)
(* Provenance of the mathematics used:                                       *)
(*   MOD_ADD_CASES / MOD_LT / MOD_REFL / EXP_ADD / LT_MULT_LCANCEL  (arith)  *)
(*   INT_REM_UNIQ / INT_OF_NUM_* / INT_POW_ADD / INT_*_DISTRIB      (int)    *)
(*   VAL_WORD_JOIN_SIMPLE / VAL_WORD_0 / VAL_BOUND / VAL_WORD_BITVAL (words) *)
(*                                                                           *)
(* Kernel trust: every SAT refutation is replayed by the LCF kernel inside   *)
(* CSAT_REFUTE (csat_lib.ml); no solver certificate is trusted and no         *)
(* compiler enters the trusted base.                                         *)
(*                                                                           *)
(* DEVELOPMENT_ONLY (not lemmas; recorded so they cannot hide):             *)
(*   BRIDGE_ADD_TAC / BRIDGE_SUB_TAC / BRIDGE_NEG_TAC are *shape dispatch*.  *)
(*   They inspect the postcondition shape (sum MOD p, difference rem p,      *)
(*   complement MOD p), split at the single conditional that shape admits,   *)
(*   and call the generic lemmas above.  The old human tail (DECARRY_RULE,   *)
(*   DESUM_RULE, BOUNDER_TAC, EQUAL_FROM_CONGRUENT_MOD_MOD,                 *)
(*   REAL_INTEGER_TAC) is not used anywhere in this file.                    *)
(* ========================================================================= *)

(* ------------------------------------------------------------------------- *)
(* 1. The modular specification, split at its single conditional correction. *)
(*                                                                           *)
(* Each lemma states the spec side in terms of the value the machine writes, *)
(* so the machine-side obligation that reaches SAT is a pure *sum* identity: *)
(* no MOD, no rem, no subtraction.                                           *)
(* ------------------------------------------------------------------------- *)

(* add: (m + n) MOD p, at most one correction because m < p and n < p. *)
let ADD_MOD_LT = prove
 (`!p m n. m < p /\ n < p /\ m + n < p ==> (m + n) MOD p = m + n`,
  REPEAT GEN_TAC THEN STRIP_TAC THEN
  SUBGOAL_THEN `(m + n) MOD p = (if m + n < p then m + n else (m + n) - p)`
    (fun th -> REWRITE_TAC[th]) THENL
   [MATCH_MP_TAC MOD_ADD_CASES THEN ASM_REWRITE_TAC[];
    ASM_REWRITE_TAC[]])
in
let ADD_MOD_GE = prove
 (`!p m n. m < p /\ n < p /\ p <= m + n ==> (m + n) MOD p = (m + n) - p`,
  REPEAT GEN_TAC THEN STRIP_TAC THEN
  SUBGOAL_THEN `(m + n) MOD p = (if m + n < p then m + n else (m + n) - p)`
    (fun th -> REWRITE_TAC[th]) THENL
   [MATCH_MP_TAC MOD_ADD_CASES THEN ASM_REWRITE_TAC[];
    SUBGOAL_THEN `~(m + n < p)` ASSUME_TAC THENL [ASM_ARITH_TAC; ALL_TAC] THEN
    ASM_REWRITE_TAC[]])
in
(* `t = (m + n) - p' in sum form, so that the residue has no subtraction *)
let ADD_GE_SUM = prove
 (`!p m n t. p <= m + n ==> ((t = (m + n) - p) <=> (t + p = m + n))`,
  REPEAT GEN_TAC THEN DISCH_TAC THEN
  MP_TAC(ISPECL [`m + n:num`; `p:num`] SUB_ADD) THEN
  ANTS_TAC THENL [ASM_REWRITE_TAC[]; DISCH_TAC] THEN
  EQ_TAC THENL
   [DISCH_THEN SUBST1_TAC THEN ASM_REWRITE_TAC[];
    DISCH_TAC THEN UNDISCH_TAC `((m + n) - p) + p = m + n` THEN
    ASM_ARITH_TAC])
in
(* the same bridge, stated once in the shape a caller applies *)
let ADD_MOD_SPLIT = prove
 (`!p m n s t.
       0 < p /\ m < p /\ n < p /\
       (m + n < p ==> s = m + n) /\
       (p <= m + n ==> t + p = m + n)
       ==> (m + n) MOD p = (if m + n < p then s else t)`,
  REPEAT GEN_TAC THEN STRIP_TAC THEN
  ASM_CASES_TAC `m + n < p` THENL
   [ASM_SIMP_TAC[ADD_MOD_LT] THEN ASM_ARITH_TAC;
    SUBGOAL_THEN `p <= m + n` ASSUME_TAC THENL [ASM_ARITH_TAC; ALL_TAC] THEN
    ASM_SIMP_TAC[ADD_MOD_GE] THEN CONV_TAC SYM_CONV THEN
    ASM_SIMP_TAC[ADD_GE_SUM] THEN ASM_ARITH_TAC])
in
(* sub: (&m - &n) rem &p, split at the borrow. *)
let SUB_REM_LT = prove
 (`!p m n. 0 < p /\ m < p /\ n < p /\ m < n
           ==> ((&m:int) - &n) rem (&p) = &((m + p) - n)`,
  REPEAT GEN_TAC THEN STRIP_TAC THEN
  SUBGOAL_THEN `n <= m + p` ASSUME_TAC THENL [ASM_ARITH_TAC; ALL_TAC] THEN
  SUBGOAL_THEN `&((m + p) - n):int = (&m + &p) - &n` ASSUME_TAC THENL
   [ASM_SIMP_TAC[INT_OF_NUM_SUB; INT_OF_NUM_ADD]; ALL_TAC] THEN
  MATCH_MP_TAC INT_REM_UNIQ THEN EXISTS_TAC `--(&1):int` THEN
  REWRITE_TAC[INT_ABS_NUM] THEN REPEAT CONJ_TAC THENL
   [UNDISCH_TAC `&((m + p) - n):int = (&m + &p) - &n` THEN CONV_TAC INT_ARITH;
    REWRITE_TAC[INT_OF_NUM_LE] THEN ARITH_TAC;
    REWRITE_TAC[INT_OF_NUM_LT] THEN ASM_ARITH_TAC])
in
let SUB_REM_GE = prove
 (`!p m n. 0 < p /\ m < p /\ n < p /\ ~(m < n)
           ==> ((&m:int) - &n) rem (&p) = &(m - n)`,
  REPEAT GEN_TAC THEN STRIP_TAC THEN
  SUBGOAL_THEN `n <= m` ASSUME_TAC THENL [ASM_ARITH_TAC; ALL_TAC] THEN
  SUBGOAL_THEN `&(m - n):int = &m - &n` ASSUME_TAC THENL
   [ASM_SIMP_TAC[INT_OF_NUM_SUB]; ALL_TAC] THEN
  MATCH_MP_TAC INT_REM_UNIQ THEN EXISTS_TAC `(&0):int` THEN
  REWRITE_TAC[INT_ABS_NUM] THEN REPEAT CONJ_TAC THENL
   [UNDISCH_TAC `&(m - n):int = &m - &n` THEN CONV_TAC INT_ARITH;
    REWRITE_TAC[INT_OF_NUM_LE] THEN ARITH_TAC;
    REWRITE_TAC[INT_OF_NUM_LT] THEN ASM_ARITH_TAC])
in
let SUB_LT_SUM = prove
 (`!p m n t. n < p /\ m < n ==> ((t = (m + p) - n) <=> (t + n = m + p))`,
  REPEAT GEN_TAC THEN STRIP_TAC THEN
  SUBGOAL_THEN `n <= m + p` ASSUME_TAC THENL [ASM_ARITH_TAC; ALL_TAC] THEN
  MP_TAC(ISPECL [`m + p:num`; `n:num`] SUB_ADD) THEN
  ANTS_TAC THENL [ASM_REWRITE_TAC[]; DISCH_TAC] THEN
  EQ_TAC THENL
   [DISCH_THEN SUBST1_TAC THEN ASM_REWRITE_TAC[];
    DISCH_TAC THEN UNDISCH_TAC `((m + p) - n) + n = m + p` THEN
    ASM_ARITH_TAC])
in
let SUB_GE_SUM = prove
 (`!m n t. ~(m < n) ==> ((t = m - n) <=> (t + n = m))`,
  REPEAT GEN_TAC THEN DISCH_TAC THEN
  SUBGOAL_THEN `n <= m` ASSUME_TAC THENL [ASM_ARITH_TAC; ALL_TAC] THEN
  MP_TAC(ISPECL [`m:num`; `n:num`] SUB_ADD) THEN
  ANTS_TAC THENL [ASM_REWRITE_TAC[]; DISCH_TAC] THEN
  EQ_TAC THENL
   [DISCH_THEN SUBST1_TAC THEN ASM_REWRITE_TAC[];
    DISCH_TAC THEN UNDISCH_TAC `(m - n) + n = m` THEN ASM_ARITH_TAC])
in
let SUB_REM_SPLIT = prove
 (`!p m n s.
       0 < p /\ m < p /\ n < p /\
       (m < n ==> s + n = m + p) /\
       (~(m < n) ==> s + n = m)
       ==> ((&m:int) - &n) rem (&p) = &s`,
  REPEAT GEN_TAC THEN STRIP_TAC THEN
  ASM_CASES_TAC `m < n` THENL
   [ASM_SIMP_TAC[SUB_REM_LT] THEN REWRITE_TAC[INT_OF_NUM_EQ] THEN
    CONV_TAC SYM_CONV THEN ASM_SIMP_TAC[SUB_LT_SUM];
    ASM_SIMP_TAC[SUB_REM_GE] THEN REWRITE_TAC[INT_OF_NUM_EQ] THEN
    CONV_TAC SYM_CONV THEN ASM_SIMP_TAC[SUB_GE_SUM]])
in
(* neg: (p - n) MOD p, split at n = 0. *)
let NEG_MOD_ZERO = prove
 (`!p n. 0 < p /\ n = 0 ==> (p - n) MOD p = 0`,
  REPEAT GEN_TAC THEN STRIP_TAC THEN ASM_REWRITE_TAC[SUB_0; MOD_REFL])
in
let NEG_MOD_NZ = prove
 (`!p n. 0 < p /\ n <= p /\ ~(n = 0) ==> (p - n) MOD p = (p - n)`,
  REPEAT GEN_TAC THEN STRIP_TAC THEN MATCH_MP_TAC MOD_LT THEN ASM_ARITH_TAC)
in
let NEG_NZ_SUM = prove
 (`!p n t. n <= p ==> ((t = p - n) <=> (t + n = p))`,
  REPEAT GEN_TAC THEN DISCH_TAC THEN
  MP_TAC(ISPECL [`p:num`; `n:num`] SUB_ADD) THEN
  ANTS_TAC THENL [ASM_REWRITE_TAC[]; DISCH_TAC] THEN
  EQ_TAC THENL
   [DISCH_THEN SUBST1_TAC THEN ASM_REWRITE_TAC[];
    DISCH_TAC THEN UNDISCH_TAC `(p - n) + n = p` THEN ASM_ARITH_TAC])
in
let NEG_MOD_SPLIT = prove
 (`!p n s.
       0 < p /\ n <= p /\
       (n = 0 ==> s = 0) /\
       (~(n = 0) ==> s + n = p)
       ==> (p - n) MOD p = s`,
  REPEAT GEN_TAC THEN STRIP_TAC THEN
  ASM_CASES_TAC `n = 0` THENL
   [ASM_SIMP_TAC[NEG_MOD_ZERO] THEN ASM_ARITH_TAC;
    ASM_SIMP_TAC[NEG_MOD_NZ] THEN CONV_TAC SYM_CONV THEN
    ASM_SIMP_TAC[NEG_NZ_SUM] THEN ASM_ARITH_TAC])
in
let BRIDGE_SPEC_LEMMAS =
  [ADD_MOD_LT; ADD_MOD_GE; ADD_GE_SUM; ADD_MOD_SPLIT;
   SUB_REM_LT; SUB_REM_GE; SUB_LT_SUM; SUB_GE_SUM; SUB_REM_SPLIT;
   NEG_MOD_ZERO; NEG_MOD_NZ; NEG_NZ_SUM; NEG_MOD_SPLIT]
in
(* ------------------------------------------------------------------------- *)
(* 2. Power-of-two scaling as a word join.                                   *)
(*                                                                           *)
(* `word_join' is free under bit-blasting (BIT_WORD_JOIN); a product by a    *)
(* power of two is instead expanded by WORDIFY_CONV into a full multiplier,  *)
(* which dominates the SAT cost.  Rewriting the shifts to joins is what      *)
(* makes the residue cheap, and it is also what removes the unconstrained-   *)
(* width problem: `word_join' has a free result type, so the width must be   *)
(* supplied here (`mk_word_ty').                                            *)
(* ------------------------------------------------------------------------- *)

let EXP2_NZ = prove
 (`!j:num. ~(2 EXP j = 0)`,
  GEN_TAC THEN REWRITE_TAC[EXP_EQ_0] THEN ARITH_TAC)
in
let LT_EXP_MULT = prove
 (`!x j k:num. x < 2 EXP k ==> 2 EXP j * x < 2 EXP (j + k)`,
  REPEAT GEN_TAC THEN DISCH_TAC THEN REWRITE_TAC[EXP_ADD] THEN
  REWRITE_TAC[LT_MULT_LCANCEL] THEN
  REWRITE_TAC[EXP_EQ_0; ARITH] THEN ASM_REWRITE_TAC[])
in
(* val(join x 0) = 2^w * val x, i.e. shifting left by a zero word. *)
let VAL_JOIN_ZERO = prove
 (`!x:M word. dimindex(:M) + dimindex(:N) = dimindex(:P)
              ==> val((word_join:M word->N word->P word) x (word 0)) =
                  2 EXP dimindex(:N) * val x`,
  REPEAT STRIP_TAC THEN
  ASM_SIMP_TAC[VAL_WORD_JOIN_SIMPLE; VAL_WORD_0; ADD_0])
in
(* the same statement one type level up, in the shape the residue has *)
let INT_JOIN_ZERO = prove
 (`!x:M word. dimindex(:M) + dimindex(:N) = dimindex(:P)
              ==> (&2:int) pow dimindex(:N) * &(val x) =
                  &(val((word_join:M word->N word->P word) x (word 0)))`,
  REPEAT STRIP_TAC THEN
  ASM_SIMP_TAC[VAL_JOIN_ZERO; GSYM INT_OF_NUM_MUL; INT_OF_NUM_POW])
in
let BITVAL_AS_VAL = prove
 (`!b. bitval b = val(word(bitval b):1 word)`,
  GEN_TAC THEN CONV_TAC SYM_CONV THEN
  MATCH_ACCEPT_TAC (INST_TYPE [`:1`, `:N`] VAL_WORD_BITVAL))
in
(* ------------------------------------------------------------------------- *)
(* 3. Conversions.                                                           *)
(* ------------------------------------------------------------------------- *)

(* real equation over integer-valued atoms -> integer equation *)
let REAL_TO_INT_CONV =
  GEN_REWRITE_CONV DEPTH_CONV
   [GSYM int_of_num_th; GSYM int_neg_th; GSYM int_add_th; GSYM int_sub_th;
    GSYM int_mul_th; GSYM int_pow_th; GSYM int_eq]
in
(* num equation -> int equation, with `&' pushed inside sum/product/power.    *)
(* INT_OF_NUM_ADD/MUL/POW are stated with `&' *outside* (`&m + &n = &(m + n)`), *)
(* so they must be used in the GSYM direction here; using them forwards folds  *)
(* the whole bignum back into a single `&(...)', which the shift-to-join       *)
(* conversion cannot see (measured: every post-simulation stage stayed SAT).   *)
let NUM_TO_INT_CONV =
  GEN_REWRITE_CONV TOP_DEPTH_CONV
   [GSYM INT_OF_NUM_EQ; GSYM INT_OF_NUM_LT; GSYM INT_OF_NUM_LE;
    GSYM INT_OF_NUM_ADD; GSYM INT_OF_NUM_MUL; GSYM INT_OF_NUM_POW]
in
(* (a + b) * c = a * c + b * c, needed because bignum_of_wordlist nests     *)
(* its shifts around a sum, not around a single digit.                       *)
let INT_DISTRIB_CONV =
  GEN_REWRITE_CONV TOP_DEPTH_CONV [INT_ADD_LDISTRIB; INT_ADD_RDISTRIB]
in
(* (&2) pow k * &(val x) -> &(val(join x 0)) at the width this x forces.     *)
(* Bottom-up application gives 2^(64*(j+1)) for j nested shifts for free.    *)
let INT_JOIN_ZERO_TH x k =
  (* `dest_word_ty' is shadowed by Library/bitmatch.ml once the ARM base is
     loaded, so the width is read off the type constructor directly. *)
  let m = Num.int_of_num
    (dest_finty(match type_of x with
                | Tyapp("word",[n]) -> n
                | _ -> failwith "INT_JOIN_ZERO_TH: not a word")) in
  let inst = INST_TYPE
    [mk_finty(Num.num_of_int m), `:M`;
     mk_finty(Num.num_of_int k), `:N`;
     mk_finty(Num.num_of_int(m + k)), `:P`] INT_JOIN_ZERO in
  let th1 = SPEC x inst in
  let cond = lhand(concl th1) in
  let side =
    prove(cond,
      CONV_TAC(LAND_CONV(BINOP_CONV DIMINDEX_CONV THENC NUM_ADD_CONV) THENC
               RAND_CONV DIMINDEX_CONV THENC NUM_EQ_CONV)) in
  (* reduce dimindex(:N) on the left so the rule is syntactically applicable.  *)
  (* The left side of the equation is `&2 pow dimindex(:N) * &(val x)'; the     *)
  (* exponent is the *left* operand of that product, so the path is            *)
  (* LAND (LAND (RAND DIMINDEX)).  The old single LAND reached `&(val x)' and   *)
  (* raised DIMINDEX_CONV, which DEPTH_CONV silently swallowed, leaving every  *)
  (* shift unconverted and every SAT call SATISFIABLE.                        *)
  CONV_RULE (LAND_CONV(LAND_CONV(RAND_CONV DIMINDEX_CONV))) (MP th1 side)
in
let INT_SHIFT_TO_JOIN_CONV : conv =
  (* the second operand is either `&(val x)' or `&(bitval b)'; the latter is    *)
  (* turned into `&(val(word(bitval b):1 word))' so both share one path.        *)
  let operand q =
    match q with
      Comb(Const("int_of_num",_),Comb(Const("val",_),x)) -> (x, REFL q)
    | Comb(Const("int_of_num",_),Comb(Const("bitval",_),_)) ->
        let bth = RAND_CONV (GEN_REWRITE_CONV I [BITVAL_AS_VAL]) q in
        (rand(rand(rand(concl bth))), bth)
    | _ -> failwith "INT_SHIFT_TO_JOIN_CONV: operand" in
  let pow2 tm =
    match tm with
      Comb(Comb(Const("int_pow",_),two),n)
        when aconv two `(&2):int` && is_numeral n -> dest_small_numeral n
    | _ -> failwith "INT_SHIFT_TO_JOIN_CONV: multiplier" in
  let mul_left p th =
    AP_TERM (mk_comb(`int_mul:int->int->int`, p)) th in
  let conv tm =
    match tm with
      Comb(Comb(Const("int_mul",_),p),q) when type_of tm = `:int` ->
        (match p with
           Comb(Comb(Const("int_pow",_),_),_) ->
             let k = pow2 p in
             let (x,bth) = operand q in
             TRANS (mul_left p bth) (INT_JOIN_ZERO_TH x k)
         | Comb(Const("int_neg",_),pt) ->
             (* accumulator equations carry the shifted flag negatively       *)
             let k = pow2 pt in
             let (x,bth) = operand q in
             let q' = rand(concl bth) in
             let bth2 = mul_left p bth in
             let negth = AP_TERM `(int_neg):int->int` (INT_JOIN_ZERO_TH x k) in
             TRANS bth2 (TRANS (SPECL [pt; q'] INT_MUL_LNEG) negth)
         | _ -> failwith "INT_SHIFT_TO_JOIN_CONV: multiplier") 
    | _ -> failwith "INT_SHIFT_TO_JOIN_CONV" in
  conv
in
let BRIDGE_SHIFTS_TO_JOINS_CONV = DEPTH_CONV INT_SHIFT_TO_JOIN_CONV
in
(* whole-term conversion from any of the three spec types to a word residue *)
let BRIDGE_TO_WORD_CONV =
  REAL_TO_INT_CONV THENC NUM_TO_INT_CONV THENC INT_DISTRIB_CONV THENC
  BRIDGE_SHIFTS_TO_JOINS_CONV
in
(* ------------------------------------------------------------------------- *)
(* 4. Case splitting on the conditionals the simulator leaves behind.        *)
(*                                                                           *)
(* The conditions may sit in the goal (the selected limb of the written      *)
(* value) or in an accumulator equation (a selected mask word).  Both are    *)
(* reduced by the same recursive split, and the condition itself stays as a *)
(* bit-blastable hypothesis, so SAT can see that the branch is consistent.   *)
(* ------------------------------------------------------------------------- *)

let bridge_conditions tm =
  map (fun t -> rand(rator(rator t))) (find_terms is_cond tm)
in
let rec BRIDGE_SPLIT_CONDITIONS_TAC (asl,w) =
  let conds = bridge_conditions w @
              List.concat (map (fun (_,th) -> bridge_conditions(concl th)) asl) in
  match conds with
  | [] -> ALL_TAC (asl,w)
  | c::_ ->
      let truerule = ASSUME c
      and falserule = EQF_INTRO (ASSUME (mk_neg c)) in
      let close rule =
        RULE_ASSUM_TAC(REWRITE_RULE[rule; COND_CLAUSES; BITVAL_CLAUSES]) THEN
        CONV_TAC(REWRITE_CONV[rule; COND_CLAUSES; BITVAL_CLAUSES]) THEN
        BRIDGE_SPLIT_CONDITIONS_TAC in
      (ASM_CASES_TAC c THENL [close truerule; close falserule]) (asl,w)
in
(* ------------------------------------------------------------------------- *)
(* 5. Eliminate the specification variables from their digit expansions.     *)
(* ------------------------------------------------------------------------- *)

let BRIDGE_HAS_VAL tm =
  mem "val" (map (fst o dest_const)
    (find_terms (fun t -> can (fst o dest_const) t) tm))
in
let rec ELIM_DIGIT_EQS_TAC (asl,w) =
  let pick = ref None in
  do_list (fun (_,th) ->
    let tm = concl th in
    if is_eq tm && type_of (lhand tm) = `:num` && not (is_var tm) then
      let (l,r) = dest_eq tm in
      if is_var l && free_in l w && not (free_in l r) then pick := Some th
      else if is_var r && free_in r w && not (free_in r l) then
        pick := Some (SYM th)) asl;
  match !pick with
  | None -> ALL_TAC (asl,w)
  | Some th -> (SUBST_ALL_TAC th THEN ELIM_DIGIT_EQS_TAC) (asl,w)
in
(* ------------------------------------------------------------------------- *)
(* 6. Certified-SAT close of the word residue.                               *)
(*                                                                           *)
(* BITBLAST_THEN consumes only word-level content; any atom it cannot expand *)
(* becomes a free proposition and the residue comes back SAT, so everything  *)
(* that is not word-level must be gone before the call.  The residue is      *)
(* presented as one closed implication so that no hypothesis is lost.        *)
(* ------------------------------------------------------------------------- *)

let BRIDGE_CSAT_TAC : tactic =
  fun (asl,w) ->
    let hyps = map (fun (_,th) -> concl th) asl in
    let imp = itlist (fun a b -> mk_imp(a,b)) hyps w in
    let th = prove(imp, CSAT_TAC) in
    (* `th' is `hyps ==> w'.  MP_TAC puts that implication in front of the goal; *)
    (* ASM_REWRITE_TAC then discharges every antecedent because each one is an    *)
    (* assumption of the goal.  (MATCH_MP_TAC peels only one implication and so   *)
    (* fails on the nested form: measured `MATCH_MP_TAC: No match'.)            *)
    (MP_TAC th THEN ASM_REWRITE_TAC[]) (asl,w)
in
(* everything that precedes the SAT call, on the goal *)
let BRIDGE_PREPARE_TAC (expand:tactic) : tactic =
  fun g ->
    (BRIDGE_SPLIT_CONDITIONS_TAC THEN
     expand THEN
     RULE_ASSUM_TAC (CONV_RULE BRIDGE_TO_WORD_CONV) THEN
     CONV_TAC BRIDGE_TO_WORD_CONV) g
in
let BRIDGE_FINISH_TAC expand : tactic =
  fun g -> (BRIDGE_PREPARE_TAC expand THEN BRIDGE_CSAT_TAC) g
in
(* ------------------------------------------------------------------------- *)
(* 7. Entry points.  `pth' is the definition theorem of the modulus, so the  *)
(* numeral replaces the constant before anything is decided.                 *)
(* ------------------------------------------------------------------------- *)

let BRIDGE_ZERO_LT_TAC p : tactic =
  SUBGOAL_THEN (mk_comb(mk_comb(`(<):num->num->bool`, `0`), p))
   (fun th -> ASSUME_TAC th) THENL [ARITH_TAC; ALL_TAC]
in
(* the modulus constant has to be replaced in the assumptions as well as in   *)
(* the goal: the range hypotheses are stated about the constant.            *)
let BRIDGE_REWRITE_MODULUS pth : tactic =
  RULE_ASSUM_TAC(REWRITE_RULE[pth]) THEN REWRITE_TAC[pth]
in
(* add: the spec is `(m + n) MOD p = <written value>' *)
let BRIDGE_ADD_TAC expand pth : tactic =
  ELIM_DIGIT_EQS_TAC THEN BRIDGE_REWRITE_MODULUS pth THEN
  (fun (asl,w) ->
    let (lhs,_) = dest_eq w in
    let (sum,p) = dest_binop `(MOD):num->num->num` lhs in
    let cond = mk_comb(mk_comb(`(<):num->num->bool`, sum), p) in
    (BRIDGE_ZERO_LT_TAC p THEN
     ASM_CASES_TAC cond THENL
      [ASM_SIMP_TAC[ADD_MOD_LT] THEN BRIDGE_FINISH_TAC expand;
       SUBGOAL_THEN (mk_comb(mk_comb(`(<=):num->num->bool`, p), sum))
         ASSUME_TAC THENL [ASM_ARITH_TAC; ALL_TAC] THEN
       ASM_SIMP_TAC[ADD_MOD_GE] THEN CONV_TAC SYM_CONV THEN
       ASM_SIMP_TAC[ADD_GE_SUM] THEN BRIDGE_FINISH_TAC expand]) (asl,w))
in
(* sub: the spec is `&<written value> = (&m - &n) rem &p' *)
let BRIDGE_SUB_TAC expand pth : tactic =
  ELIM_DIGIT_EQS_TAC THEN BRIDGE_REWRITE_MODULUS pth THEN
  (fun (asl,w) ->
    let (lhs,rhs) = dest_eq w in
    let (diff,p) = dest_binop `(rem):int->int->int` rhs in
    let (m,n) = dest_binop `(-):int->int->int` diff in
    let cond = mk_comb(mk_comb(`(<):num->num->bool`, rand m), rand n) in
    (BRIDGE_ZERO_LT_TAC (rand p) THEN
     ASM_CASES_TAC cond THENL
      [ASM_SIMP_TAC[SUB_REM_LT] THEN REWRITE_TAC[INT_OF_NUM_EQ] THEN
       ASM_SIMP_TAC[SUB_LT_SUM] THEN BRIDGE_FINISH_TAC expand;
       ASM_SIMP_TAC[SUB_REM_GE] THEN REWRITE_TAC[INT_OF_NUM_EQ] THEN
       ASM_SIMP_TAC[SUB_GE_SUM] THEN BRIDGE_FINISH_TAC expand])
    (asl,w))
in
(* neg: the spec is `&<written value> = &((p - n) MOD p)' *)
let BRIDGE_NEG_TAC expand pth : tactic =
  ELIM_DIGIT_EQS_TAC THEN BRIDGE_REWRITE_MODULUS pth THEN
  CONV_TAC REAL_TO_INT_CONV THEN
  (fun (asl,w) ->
    let (_,rhs) = dest_eq w in
    let modtm = rand rhs in
    let (diff,p) = dest_binop `(MOD):num->num->num` modtm in
    let n = rand diff in
    let cond = mk_comb(mk_comb(`(=):num->num->bool`, n), `0`) in
    (BRIDGE_ZERO_LT_TAC p THEN
     ASM_CASES_TAC cond THENL
      [ASM_SIMP_TAC[NEG_MOD_ZERO] THEN BRIDGE_FINISH_TAC expand;
       ASM_SIMP_TAC[NEG_MOD_NZ] THEN
       ASM_SIMP_TAC[INT_OF_NUM_SUB] THEN BRIDGE_FINISH_TAC expand])
    (asl,w))
in
(fun g ->
  try
    (MAP_EVERY X_GEN_TAC
 [`z:int64`; `x:int64`; `y:int64`; `m:num`; `n:num`; `pc:num`] THEN
REWRITE_TAC[C_ARGUMENTS; C_RETURN; SOME_FLAGS; NONOVERLAPPING_CLAUSES] THEN
DISCH_THEN(REPEAT_TCL CONJUNCTS_THEN ASSUME_TAC) THEN
REWRITE_TAC[BIGNUM_FROM_MEMORY_BYTES] THEN ENSURES_INIT_TAC "s0" THEN
BIGNUM_DIGITIZE_TAC "m_" `read (memory :> bytes (x,8 * 4)) s0` THEN
BIGNUM_DIGITIZE_TAC "n_" `read (memory :> bytes (y,8 * 4)) s0` THEN
ARM_ACCSTEPS_TAC BIGNUM_ADD_P256_EXEC [3;4;7;8;10;12;13;15;16] (1--22) THEN
ENSURES_FINAL_STATE_TAC THEN ASM_REWRITE_TAC[] THEN STRIP_TAC THEN
RULE_ASSUM_TAC(REWRITE_RULE [REAL_BITVAL_NOT; ADD_CLAUSES; VAL_WORD_BITVAL]) THEN
CONV_TAC SYM_CONV THEN CONV_TAC(RAND_CONV BIGNUM_LEXPAND_CONV) THEN
ASM_REWRITE_TAC[] THEN DISCARD_STATE_TAC "s22" THEN
BRIDGE_ADD_TAC (CONV_TAC (ONCE_DEPTH_CONV BIGNUM_OF_WORDLIST_CONV)) p_256) g
  with e ->

    raise e)
)
